// ========================================================= // Geo-Checker / Quake-Checker 統合マップ・地震ロガー機能 // ========================================================= // グローバル変数の定義 let map; let baseLayers = {}; let overlayLayers = {}; let geologyTileLayer; let earthquakeLayerGroup = null; // 地震レイヤーの管理用 // --------------------------------------------------- // 1. マップ初期化 (産総研地質図の乗算合成・CORS対応版) // --------------------------------------------------- function initMap() { // Leaflet Canvas レンダラーを適用して描画を高速化 map = L.map('map', { zoomControl: false, renderer: L.canvas() }).setView([32.789, 130.705], 8); // 熊本中心 L.control.zoom({ position: 'bottomright' }).addTo(map); // 地質図専用のPane（描画レイヤー階層）を生成し、背景地図(200)の上に固定(400) map.createPane('geologyPane'); map.getPane('geologyPane').style.zIndex = 400; map.getPane('geologyPane').style.pointerEvents = 'none'; // クリック操作を下のマップへ透過 map.getPane('geologyPane').style.mixBlendMode = 'multiply'; // ★乗算モード（下の道路や文字が透ける） // ① 国土地理院 標準地図 const stdMap = L.tileLayer('https://cyberjapandata.gsi.go.jp/xyz/std/{z}/{x}/{y}.png', { maxZoom: 18, crossOrigin: true, // 画像保存機能のためのCORS設定 attribution: '国土地理院' }).addTo(map); // ② 国土地理院 陰影起伏図 const hillShadeMap = L.tileLayer('https://cyberjapandata.gsi.go.jp/xyz/hillshademap/{z}/{x}/{y}.png', { maxZoom: 16, crossOrigin: true, attribution: '国土地理院' }); // ③ 産総研 シームレス地質図V2 geologyTileLayer = L.tileLayer('https://gbank.gsj.jp/seamless/v2/api/1.2.1/tiles/{z}/{y}/{x}.png', { pane: 'geologyPane', // 専用Paneに強制配置 minZoom: 5, maxNativeZoom: 13, maxZoom: 18, opacity: 0.75, // UI側で透過度変更可能 crossOrigin: true, attribution: '産総研地質調査総合センター' }).addTo(map); baseLayers = { "地理院 標準地図": stdMap, "地理院 陰影起伏図 (地形)": hillShadeMap }; overlayLayers = { "産総研 シームレス地質図V2": geologyTileLayer }; L.control.layers(baseLayers, overlayLayers, { position: 'topright' }).addTo(map); } // --------------------------------------------------- // 2. 気象庁API 連携処理 (JMA地震データ取得・解析) // --------------------------------------------------- /** * 気象庁独特の座標文字列を 緯度・経度・深さ(km) に変換するパーサー * 例: "+32.8+130.7-10000/" -> lat: 32.8, lon: 130.7, depth: 10 */ function parseJMACoordinate(cod) { if (!cod) return null; // 正規表現で 緯度, 経度, 深さ を抽出 const match = cod.match(/^([+-]\d+(?:\.\d+)?)([+-]\d+(?:\.\d+)?)(?:([+-]\d+(?:\.\d+)?))?\//); if (!match) return null; return { lat: parseFloat(match[1]), lon: parseFloat(match[2]), // JMAの深さはメートル(負の値)で来るため、正の数のkmに変換 depth: match[3] ? Math.abs(parseFloat(match[3])) / 1000 : 0 }; } /** * 気象庁APIから最新の地震履歴を取得しマップに描画 * @param {number} minMagnitude - フィルタリングする最小マグニチュード * @param {number} maxDepth - フィルタリングする最大震源深さ(km) */ async function fetchJMAEarthquakes(minMagnitude = 2.0, maxDepth = 300) { const url = 'https://www.jma.go.jp/bosai/quake/data/list.json'; try { const response = await fetch(url); if (!response.ok) throw new Error("気象庁API通信エラー"); const data = await response.json(); const features = []; data.forEach(eq => { if (!eq.cod) return; // 座標がないデータはスキップ const coords = parseJMACoordinate(eq.cod); if (!coords) return; const mag = parseFloat(eq.mag) || 0; // ★ UIで設定された「最小マグニチュード」かつ「最大震源深さ」でフィルタリング if (mag >= minMagnitude && coords.depth <= maxDepth) { features.push({ "type": "Feature", "geometry": { "type": "Point", "coordinates": [coords.lon, coords.lat, coords.depth] }, "properties": { "title": eq.anm || "震源地不明", "magnitude": mag, "depth": coords.depth, "maxInt": eq.maxi || "不明", "time": eq.at || eq.en_dt // 発生日時 } }); } }); const geoJsonData = { "type": "FeatureCollection", "features": features }; // マップへプロット実行 plotGeoJSON(geoJsonData); } catch (error) { console.error("気象庁地震データの取得に失敗しました:", error); alert("気象庁APIからのデータ取得に失敗しました。"); } } // --------------------------------------------------- // 3. マップへのプロットとUI (GeoJSON描画) // --------------------------------------------------- function plotGeoJSON(geoJsonData) { // 既存の地震レイヤーがあれば削除（リセットして再描画するため） if (earthquakeLayerGroup) { map.removeLayer(earthquakeLayerGroup); } earthquakeLayerGroup = L.geoJSON(geoJsonData, { pointToLayer: function (feature, latlng) { const mag = feature.properties.magnitude; // マグニチュードに応じた円の大きさと色 const radius = Math.max(mag * 2.5, 3); let fillColor = "#ffcc00"; // M3未満（黄色） if (mag >= 5.0) fillColor = "#cc0000"; // M5以上（赤色） else if (mag >= 3.0) fillColor = "#ff6600"; // M3〜M5未満（オレンジ） return L.circleMarker(latlng, { radius: radius, fillColor: fillColor, color: "#000", weight: 1, opacity: 1, fillOpacity: 0.8 }); }, onEachFeature: function (feature, layer) { const props = feature.properties; const maxInt = props.maxInt ? `震度 ${props.maxInt}` : "不明"; // 見やすいポップアップのHTML構成（震度を赤文字で強調） const popupContent = `

${props.title}

最大震度: ${maxInt}
規模 (M): ${props.magnitude}
深さ: ${props.depth} km
発生日時: ${new Date(props.time).toLocaleString('ja-JP')}
`; layer.bindPopup(popupContent); } }); // マップにレイヤーを追加 earthquakeLayerGroup.addTo(map); } // --------------------------------------------------- // 4. アプリケーション初期化実行 // --------------------------------------------------- window.onload = initMap;
